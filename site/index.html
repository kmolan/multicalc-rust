Autodiff

Write it once. Differentiate it exactly.

The same formula runs at f64 and at a dual number. No step size, no truncation error.

let f = scalar_fn!(|x| x * x * x - constant(2.0) * x);    // x³ − 2x

let slope = derivative(&f, 2.0);                          // 10, exact
let root  = Newton::new().solve(&f, 2.0)?.root;           // √2
let area  = integral(&|x: f64| f.eval(x), [0.0, 2.0])?;   // 0

let mut filter = KalmanFilter::new(state, covariance, model);
filter.predict();
filter.update(measurement)?;                              // 34 ns, no heap
